local function loadConfig()
    if isfile and isfile(CONFIG_FILE) then
        local ok, result = pcall(function()
            return HttpService:JSONDecode(readfile(CONFIG_FILE))
        end)
        if ok and type(result) == "table" then
            if type(result.Keybind) == "string"         then config.Keybind = result.Keybind end
            if type(result.AntiBatEnabled) == "boolean" then config.AntiBatEnabled = result.AntiBatEnabled end
            if type(result.InfJumpEnabled) == "boolean" then config.InfJumpEnabled = result.InfJumpEnabled end
            if type(result.AntiDieEnabled) == "boolean" then config.AntiDieEnabled = result.AntiDieEnabled end
            if type(result.BackgroundId) == "string"    then config.BackgroundId = result.BackgroundId end
            if type(result.MenuScale) == "number"       then config.MenuScale = result.MenuScale end
            if type(result.Locked) == "boolean"         then config.Locked = result.Locked end
            if type(result.AntiBatMode) == "number"      then config.AntiBatMode = math.clamp(math.floor(result.AntiBatMode), 1, 2) end
            if result.Theme == "Classic" or result.Theme == "Halloween" then config.Theme = result.Theme end
        end
    end
    for i, id in ipairs(BACKGROUND_IDS) do
        if id == config.BackgroundId then bgIndex = i break end
    end
end

-- ─── Anti-Bat ────────────────────────────────────────────────────────────────
local antiBatConn   = nil
local antiBatActive = false

local ANTI_MODE_NAMES = {
    [1] = "V1 Normal",
    [2] = "V2 Orbital",
}

local function startAntiBat()
    if antiBatConn then return end
    antiBatActive = true
    local phase = 0
    antiBatConn = RunService.Heartbeat:Connect(function(dt)
        if not antiBatActive then return end
        local character = localPlayer.Character
        if not character then return end
        local hrp = character:FindFirstChild("HumanoidRootPart")
        if not hrp then return end
        local v = hrp.AssemblyLinearVelocity or hrp.Velocity
        local mode = config.AntiBatMode or 1
        phase = phase + (dt or 0.016)

        if mode == 1 then
            hrp.AssemblyLinearVelocity = Vector3.new(
                math.random(-ANTIBAT_RANGE, ANTIBAT_RANGE),
                v.Y,
                math.random(-ANTIBAT_RANGE, ANTIBAT_RANGE)
            )
            RunService.RenderStepped:Wait()
            if hrp and hrp.Parent then
                hrp.AssemblyLinearVelocity = v
            end
        elseif mode == 2 then
            local r = ANTIBAT_RANGE * 0.7
            local ox = math.cos(phase * 16) * r
            local oz = math.sin(phase * 16) * r
            hrp.AssemblyLinearVelocity = Vector3.new(ox, v.Y, oz)
            RunService.RenderStepped:Wait()
            if hrp and hrp.Parent then
                hrp.AssemblyLinearVelocity = v
            end
        else
            hrp.AssemblyLinearVelocity = Vector3.new(
                math.random(-ANTIBAT_RANGE, ANTIBAT_RANGE),
                v.Y,
                math.random(-ANTIBAT_RANGE, ANTIBAT_RANGE)
            )
            RunService.RenderStepped:Wait()
            if hrp and hrp.Parent then
                hrp.AssemblyLinearVelocity = v
            end
        end
    end)
end

local function stopAntiBat()
    antiBatActive = false
    if antiBatConn then
        antiBatConn:Disconnect()
        antiBatConn = nil
    end
end

-- ─── Inf Jump (Hold) ─────────────────────────────────────────────────────────
local infJumpConn   = nil
local infJumpActive = false

local function applyInfJumpBoost(root)
    if not root then return end
    local velocity = root.AssemblyLinearVelocity
    if velocity.Y < INF_JUMP_MIN_Y then
        root.AssemblyLinearVelocity = Vector3.new(velocity.X, INF_JUMP_BOOST, velocity.Z)
    end
    if velocity.Y < INF_JUMP_MAX_FALL then
        root.AssemblyLinearVelocity = Vector3.new(velocity.X, INF_JUMP_MAX_FALL, velocity.Z)
    end
end

local function startInfJump()
    if infJumpConn then return end
    infJumpActive = true
    config.InfJumpEnabled = true
    infJumpConn = RunService.PreSimulation:Connect(function()
        if not infJumpActive then return end
        local char = localPlayer.Character
        if not char then return end
        local hum  = char:FindFirstChildOfClass("Humanoid")
        local root = char:FindFirstChild("HumanoidRootPart")
        if not hum or not root then return end
        local held = UserInputService:IsKeyDown(Enum.KeyCode.Space)
            or UserInputService:IsKeyDown(Enum.KeyCode.ButtonA)
            or (hum.Jump == true)
        if held then
            applyInfJumpBoost(root)
        else
            local v = root.AssemblyLinearVelocity
            if v.Y < INF_JUMP_MAX_FALL then
                root.AssemblyLinearVelocity = Vector3.new(v.X, INF_JUMP_MAX_FALL, v.Z)
            end
        end
    end)
end

local function stopInfJump()
    infJumpActive = false
    config.InfJumpEnabled = false
    if infJumpConn then
        infJumpConn:Disconnect()
        infJumpConn = nil
    end
end

-- ─── Anti Die + Anti Fling (Integrado) ────────────────────────────────────────
local antiDieConn   = nil
local antiDieActive = false
local flingConn     = nil
local flingActive   = false

local function startAntiDie()
    if antiDieConn then return end
    antiDieActive = true
    config.AntiDieEnabled = true
    
    antiDieConn = RunService.Heartbeat:Connect(function()
        if not antiDieActive then return end
        local char = localPlayer.Character
        if not char then return end
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum and hum.Health < hum.MaxHealth then
            hum.Health = hum.MaxHealth
        end
    end)
    
    if not flingConn then
        flingActive = true
        flingConn = RunService.Heartbeat:Connect(function()
            if not flingActive then return end
            local char = localPlayer.Character
            if not char then return end
            local root = char:FindFirstChild("HumanoidRootPart")
            if not root then return end
            local v = root.AssemblyLinearVelocity
            if v.Magnitude > 120 then
                root.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
                root.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
            end
        end)
    end
    
    pcall(function()
        game:GetService("StarterGui"):SetCore("ResetButtonCallback", false)
    end)
end

local function stopAntiDie()
    antiDieActive = false
    config.AntiDieEnabled = false
    if antiDieConn then
        antiDieConn:Disconnect()
        antiDieConn = nil
    end
    if flingConn then
        flingConn:Disconnect()
        flingConn = nil
    end
    flingActive = false
    pcall(function()
        game:GetService("StarterGui"):SetCore("ResetButtonCallback", true)
    end)
end

local HW_EMOJIS = {"🎃", "👻", "🦇", "🕷", "💀", "🕸", "🍬", "🧡", "🌙", "✨", "🧛", "🧟"}
local hwParticles = {}
local hwConn = nil

local function clearHalloweenFX()
    if hwConn then
        hwConn:Disconnect()
        hwConn = nil
    end
    for _, p in ipairs(hwParticles) do
        if p and p.Parent then p:Destroy() end
    end
    for i = #hwParticles, 1, -1 do hwParticles[i] = nil end
    halloweenLayer.Visible = false
end

local function spawnHalloweenFX()
    clearHalloweenFX()
    halloweenLayer.Visible = true
    local count = 14
    for i = 1, count do
        local lbl = Instance.new("TextLabel")
        lbl.BackgroundTransparency = 1
        lbl.Size = UDim2.fromOffset(18, 18)
        lbl.Text = HW_EMOJIS[((i - 1) % #HW_EMOJIS) + 1]
        lbl.TextSize = ({14, 16, 18, 12, 20})[((i - 1) % 5) + 1]
        lbl.ZIndex = 4
        lbl.Rotation = math.random(-20, 20)
        lbl.TextTransparency = 0.15 + (i % 4) * 0.08
        lbl.Parent = halloweenLayer
        local data = {
            gui = lbl,
            x = math.random() * 0.9 + 0.05,
            y = math.random() * 0.9 + 0.05,
            sp = 0.08 + math.random() * 0.18,
            amp = 0.02 + math.random() * 0.04,
            phase = math.random() * math.pi * 2,
            rotSp = (math.random() - 0.5) * 40,
            bob = 0.6 + math.random() * 1.2,
        }
        lbl.Position = UDim2.new(data.x, 0, data.y, 0)
        table.insert(hwParticles, data)
    end
    local t0 = os.clock()
    hwConn = RunService.RenderStepped:Connect(function()
        if not halloweenLayer.Visible then return end
        local t = os.clock() - t0
        for _, d in ipairs(hwParticles) do
            if d.gui and d.gui.Parent then
                local ny = d.y + math.sin(t * d.bob + d.phase) * d.amp
                local nx = d.x + math.cos(t * d.sp * 0.7 + d.phase) * (d.amp * 0.6)
                d.y = d.y - d.sp * 0.0035
                if d.y < -0.05 then
                    d.y = 1.05
                    d.x = math.random() * 0.9 + 0.05
                    d.gui.Text = HW_EMOJIS[math.random(1, #HW_EMOJIS)]
                end
                d.gui.Position = UDim2.new(math.clamp(nx, 0, 0.95), 0, math.clamp(ny, -0.05, 1.05), 0)
                d.gui.Rotation = d.gui.Rotation + d.rotSp * 0.016
            end
        end
    end)
end

    -- Toggle

    track.Position            = UDim2.new(1, -12, 0.5, 0)

    track.Parent              = card
    Instance.new("UICorner", track).CornerRadius = UDim.new(1, 0)

    local knob = Instance.new("Frame")
    knob.AnchorPoint      = Vector2.new(0, 0.5)
    knob.Position         = initialState and UDim2.new(1, -(knobS + 3), 0.5, 0) or UDim2.new(0, 3, 0.5, 0)
    knob.Size             = UDim2.fromOffset(knobS, knobS)
    knob.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
    knob.BorderSizePixel  = 0
    knob.ZIndex           = 8
    knob.Parent           = track
    Instance.new("UICorner", knob).CornerRadius = UDim.new(1, 0)

    local state = initialState
    local function setVisual(on)
        state = on
        TweenService:Create(track, TweenInfo.new(0.2), {
            BackgroundColor3 = on and theme.accentCyan or theme.toggleOff
        }):Play()
        TweenService:Create(knob, TweenInfo.new(0.2), {
            Position = on and UDim2.new(1, -(knobS + 3), 0.5, 0) or UDim2.new(0, 3, 0.5, 0)

            Color = on and theme.accentCyan or theme.soft,

        }):Play()
        TweenService:Create(card, TweenInfo.new(0.2), {
            BackgroundTransparency = on and 0.35 or 0.5
        }):Play()
    end

    btn.Size                   = UDim2.new(1, 0, 1, 0)

    btn.Text                   = ""

    end)
    btn.MouseLeave:Connect(function()
        TweenService:Create(card, TweenInfo.new(0.15), { BackgroundTransparency = state and 0.35 or 0.5 }):Play()
    end)
    btn.MouseButton1Click:Connect(function()
        setVisual(not state)
        onToggle(state)
    end)

    return setVisual, cardH
end

    for _, id in ipairs(BACKGROUND_IDS) do

            stillValid = true

        end
    end
    if not stillValid then
        config.BackgroundId = BACKGROUND_IDS[1]
    end
    
    bgGradientFrame.BackgroundColor3 = theme.bg
    bgGradient.Color = theme.bgGradient
    settingsGradient.Color = theme.bgGradient

        if btn then
            btn.BackgroundColor3 = theme.soft
            btn.TextColor3 = theme.white
        end
    end
    
    if background then
        background.Image = "rbxassetid://" .. tostring(config.BackgroundId)
    end
    if bgCells then
        for i, cell in ipairs(bgCells) do
            local id = BACKGROUND_IDS[i]
            if cell and id then
                cell.Image = "rbxassetid://" .. tostring(id)
                local stroke = cell:FindFirstChildOfClass("UIStroke")
                if stroke then
                    local selected = tostring(id) == tostring(config.BackgroundId)
                    stroke.Color = selected and theme.accentCyan or Color3.fromRGB(60, 60, 70)
                    stroke.Transparency = selected and 0.1 or 0.6
                end
            end
        end
    end
    if refreshThemeBtns then refreshThemeBtns() end
    if config.Theme == "Halloween" then
        spawnHalloweenFX()
    else
        clearHalloweenFX()
    end
    saveConfig()
    showToast((config.Theme == "Halloween" and "🎃 Theme: Halloween" or "Theme: Classic"), true)
end

-- ─── DRAG ──────────────────────────────────────────────────────────────────
local dragging   = false
local dragStart  = nil
local startPos   = nil
local touchIndex = nil

local function clampToViewport(pos)
    local cam = workspace.CurrentCamera
    local viewport = cam and cam.ViewportSize or Vector2.new(1280, 720)
    local inset = GuiService:GetGuiInset()
    local x = math.clamp(pos.X.Offset + pos.X.Scale * viewport.X, 0,
        math.max(0, viewport.X - frame.AbsoluteSize.X))
    local y = math.clamp(pos.Y.Offset + pos.Y.Scale * (viewport.Y - inset.Y), 0,
        math.max(0, viewport.Y - inset.Y - frame.AbsoluteSize.Y))
    return UDim2.fromOffset(x, y)
end

local function beginDrag(input)
    if config.Locked then return end
    if settingsOpen then return end
    if input.UserInputType == Enum.UserInputType.MouseButton1
    or input.UserInputType == Enum.UserInputType.Touch then
        dragging   = true
        dragStart  = input.Position
        startPos   = frame.Position
        touchIndex = input.UserInputType == Enum.UserInputType.Touch and input.UserInputIndex or nil
    end
end

frame.InputBegan:Connect(beginDrag)
header.InputBegan:Connect(beginDrag)
if dragCatch then
    dragCatch.InputBegan:Connect(beginDrag)
end
contentScroll.InputBegan:Connect(beginDrag)

UserInputService.InputChanged:Connect(function(input)
    if not dragging or config.Locked then return end
    local isTouch = input.UserInputType == Enum.UserInputType.Touch
    if isTouch or input.UserInputType == Enum.UserInputType.MouseMovement then
        if isTouch and touchIndex and input.UserInputIndex ~= touchIndex then return end
        local delta = input.Position - dragStart
        frame.Position = clampToViewport(UDim2.new(
            startPos.X.Scale, startPos.X.Offset + delta.X,
            startPos.Y.Scale, startPos.Y.Offset + delta.Y))
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if not dragging then return end
    local isTouch = input.UserInputType == Enum.UserInputType.Touch
    if isTouch or input.UserInputType == Enum.UserInputType.MouseButton1 then
        if isTouch and touchIndex and input.UserInputIndex ~= touchIndex then return end
        dragging   = false
        touchIndex = nil
    end
end)
